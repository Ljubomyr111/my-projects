What is Big-O complexity in this algorithm?
int f1(int arr[], int n, int target) { 
   for (int i = 0; i < n; i++) { 
       if (arr[i] == target) { 
           return i; 
       } 
   } 
   return -1; 
}

Answer
Time complexity: O(n) - this linear search algorithm
Space complexity: O(1)

What is Big-O complexity in this algorithm?
int f2(int arr[], int n, int target) { 
   int low = 0, high = n - 1; 
   while (low <= high) { 
       int mid = (low + high) / 2; 
       if (arr[mid] == target) { 
           return mid;  
       } else if (arr[mid] < target) { 
           low = mid + 1;  // Search in the right half 
       } else { 
           high = mid - 1;  // Search in the left half 
       } 
   } 
   return -1;  // Target not found 
}

Answer
Time complexity: O(log(n)) - this binary search algorithm
Space complexity: O(1)

What is Big-O complexity in this algorithm?
void f3(int arr[], int n) { 
   for (int i = 0; i < n - 1; i++) { 
       for (int j = 0; j < n - i - 1; j++) { 
           if (arr[j] > arr[j + 1]) { 
               // Swap arr[j] and arr[j + 1] 
               int temp = arr[j]; 
               arr[j] = arr[j + 1]; 
               arr[j + 1] = temp; 
           } 
       } 
   } 
}

Answer
Time complexity: O(n2) - this is bubble sort algorithm
Space complexity: O(1)

}

Answer

Space complexity: O(1)

What is Big-O complexity in this algorithm?
void f5(int mat1[N][N], int mat2[N][N], int result[N][N]) { 
   for (int i = 0; i < N; i++) { 
       for (int j = 0; j < N; j++) { 
           result[i][j] = 0; 
           for (int k = 0; k < N; k++) { 
               result[i][j] += mat1[i][k] * mat2[k][j]; 
           } 
       } 
   } 
}

Answer
Time complexity: O(n3) - two arrays multiplication
Space complexity: O(1)

What is Big-O complexity in this algorithm?
int fibonacci(int n) { 
   if (n <= 1) { 
       return n; 
   } 
   return fibonacci(n - 1) + fibonacci(n - 2); 
}

Answer
O(2n) - fibonacci numbers
